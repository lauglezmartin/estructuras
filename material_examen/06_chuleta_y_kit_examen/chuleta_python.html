6Clases y POO ↑ índice

Clase completa (molde del examen)

class PersonajeTeletubby:
    """Representa a un Teletubby."""

    saludo_oficial = "¡Tubby abrazo!"     # atributo de CLASE

    def __init__(self, nombre, color, antena, objeto_favorito):
        """Constructor: atributos de INSTANCIA."""
        self.nombre = nombre                # público
        self.color = color
        self.antena = antena
        self.objeto_favorito = objeto_favorito
        self.__nivel_hambre = 50            # privado (__)
        self.__nivel_felicidad = 100

    def jugar(self):
        """Juega: sube hambre y felicidad."""
        print(f"{self.nombre} juega con {self.objeto_favorito}")
        self.__nivel_hambre += 20
        self.__nivel_felicidad += 10

    def ver_estado(self):                   # getter
        """Devuelve el estado como texto."""
        return (f"Felicidad: {self.__nivel_felicidad} | "
                f"Hambre: {self.__nivel_hambre}")

    def __str__(self):                      # para el usuario
        # formato del enunciado: Teletubby Tinky Winky (Morado) - Feliz: 100
        return (f"Teletubby {self.nombre} ({self.color}) - "
                f"Feliz: {self.__nivel_felicidad}")

    def __repr__(self):                     # técnico
        return (f"PersonajeTeletubby('{self.nombre}', '{self.color}', "
                f"'{self.antena}', '{self.objeto_favorito}')")

Getters / setters con validación

class Producto:
    """Producto con precio privado."""

    def __init__(self, nombre, precio):
        self.nombre = nombre
        self.__precio = precio

    def get_precio(self):
        """Getter: devuelve el precio."""
        return self.__precio

    def set_precio(self, nuevo):
        """Setter: solo acepta precios positivos."""
        if nuevo <= 0:
            print("Precio no válido")
            return
        self.__precio = nuevo

class Tienda:
    """Gestiona una lista de productos."""

    def __init__(self):
        self.productos = []                     # lista vacía

    def anadir(self, producto):
        self.productos.append(producto)

    def listar(self):
        for i, p in enumerate(self.productos, start=1):
            print(f"{i}. {p}")

    def buscar(self, nombre):
        for p in self.productos:
            if p.nombre.lower() == nombre.lower():
                return p
        return None                             # fuera del for

    def eliminar(self, nombre):
        p = self.buscar(nombre)
        if p is None:
            return False
        self.productos.remove(p)
        return True
# Filtrar (comprensión)
baratos = [p for p in tienda.productos if p.get_precio() < 10]

# Ordenar sin cambiar la original
por_precio = sorted(tienda.productos, key=lambda p: p.get_precio())
# Ordenar la propia lista por nombre, de la Z a la A
tienda.productos.sort(key=lambda p: p.nombre, reverse=True)

# Mayor / menor por un atributo
caro = max(tienda.productos, key=lambda p: p.get_precio())

# Total sumando un atributo

# Lista de nombres

9Comprensiones y lambda ↑ índice

Comprensión de listas

#        [EXPRESIÓN  for ELEM in ITERABLE  if FILTRO]
numeros = [1, 2, 3, 4, 5]
cubos = [n ** 3 for n in numeros]              # [1, 8, 27, 64, 125]
pares = [n for n in numeros if n % 2 == 0]     # [2, 4]
largas = [p for p in palabras if len(p) > 3]
etiquetas = ["par" if n % 2 == 0 else "impar" for n in numeros]

Comprensión de diccionarios

cuadrados = {n: n ** 2 for n in range(5)}
# {0: 0, 1: 1, 2: 4, 3: 9, 4: 16}
caros = {k: v for k, v in precios.items() if v > 100}

Equivale a un for + append, pero en una línea:

cubos = []
for n in numeros:
    cubos.append(n ** 3)

lambda = función anónima de una línea

doble = lambda x: x * 2            # lambda PARAMS: EXPRESIÓN
doble(4)                           # 8

# map: transforma cada elemento
cuadrados = list(map(lambda x: x ** 2, numeros))

# filter: se queda con los que cumplen
pares = list(filter(lambda x: x % 2 == 0, numeros))

# sorted / sort / max / min con key
sorted(palabras, key=lambda p: len(p))
max(alumnos, key=lambda a: a["nota"])
Si me piden…Uso…
Nueva lista transformada "en una línea"[f(x) for x in lista]
Filtrar "en una línea"[x for x in lista if cond]
Usar map() con lambdalist(map(lambda x: ..., lista))
Usar filter()list(filter(lambda x: cond, lista))
Ordenar por un criteriosorted(lista, key=lambda x: ...)
★ FORMA QUE EXIGE EL PROFESOR
  • Cubos y filtrado de palabras: comprensión de listas en una sola línea (un for + append funciona, pero no es lo que pide).
  • Cuadrados con list(map(lambda x: x**2, numeros)) (una comprensión funciona, pero no es lo que pide).
✗ Errores típicos
  • Olvidar list() alrededor de map/filter: imprime <map object at ...>.
  • Poner return en una lambda → SyntaxError.
  • El if de filtro va al final; el if/else de transformar va al principio.
